# libFuzzer targets. Clang only: libFuzzer ships with the Clang toolchain.
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR
            "JSONV_BUILD_FUZZERS requires Clang -- libFuzzer ships with the Clang toolchain. "
            "Re-run with -DCMAKE_CXX_COMPILER=clang++"
           )
endif()

# The fuzzers link a second, instrumented copy of the library rather than `jsonv` itself. Linking the normal target
# would leave the code under test uninstrumented: libFuzzer would get no coverage feedback to guide mutation, and
# AddressSanitizer would only see allocations, not the loads and stores inside the parser. Building separately also
# keeps the installed target and its export set free of sanitizer flags.
#
# The source list is read back from the real target so there is one statement of what the library is made of.
# SOURCES holds the paths as src/jsonv/CMakeLists.txt wrote them -- relative to that directory, with the headers it
# lists for IDEs alongside -- so keep the .cpp files and make each absolute against jsonv's own directory.
get_target_property(jsonv_source_dir jsonv SOURCE_DIR)
get_target_property(jsonv_fuzz_sources jsonv SOURCES)
list(FILTER jsonv_fuzz_sources INCLUDE REGEX "\\.cpp$")
list(TRANSFORM jsonv_fuzz_sources PREPEND "${jsonv_source_dir}/" REGEX "^[^/]")
list(LENGTH jsonv_fuzz_sources jsonv_fuzz_source_count)
if(jsonv_fuzz_source_count EQUAL 0)
    message(FATAL_ERROR "No library sources found on target jsonv")
endif()

add_library(jsonv-fuzz-lib STATIC ${jsonv_fuzz_sources})
target_include_directories(jsonv-fuzz-lib
                           PUBLIC
                             "${PROJECT_SOURCE_DIR}/include"
                           PRIVATE
                             "${PROJECT_SOURCE_DIR}/src"
                          )
target_compile_features(jsonv-fuzz-lib PUBLIC cxx_std_23)
# The headers key `JSONV_PUBLIC` off JSONV_SO, so the library and the harnesses have to agree on it.
target_compile_definitions(jsonv-fuzz-lib
                           PUBLIC
                             JSONV_SO=0
                           PRIVATE
                             JSONV_COMPILING=1
                          )
target_compile_options(jsonv-fuzz-lib
                       PUBLIC
                         ${JSONV_SANITIZER_FLAGS}
                       PRIVATE
                         # Coverage instrumentation without libFuzzer's `main`, which belongs to the executables.
                         -fsanitize=fuzzer-no-link
                      )
target_link_options(jsonv-fuzz-lib PUBLIC ${JSONV_SANITIZER_FLAGS})

foreach(fuzzer IN ITEMS parse roundtrip)
    add_executable(jsonv-fuzz-${fuzzer} "${fuzzer}_fuzz.cpp")
    target_link_libraries(jsonv-fuzz-${fuzzer} PRIVATE jsonv-fuzz-lib)
    target_compile_options(jsonv-fuzz-${fuzzer} PRIVATE -fsanitize=fuzzer)
    target_link_options(jsonv-fuzz-${fuzzer} PRIVATE -fsanitize=fuzzer)
endforeach()

# Interprocedural optimization interacts badly with the sanitizer and coverage instrumentation, and fuzzing does not
# need it. JSONV_LTO is a property of the `jsonv` target alone, so there is nothing of this project's to undo here;
# this guards against a user-level CMAKE_INTERPROCEDURAL_OPTIMIZATION. The per-configuration properties have to be
# cleared alongside the plain one: CMake reads `INTERPROCEDURAL_OPTIMIZATION_<CONFIG>` in preference to
# `INTERPROCEDURAL_OPTIMIZATION` when it is set.
set_target_properties(jsonv-fuzz-lib jsonv-fuzz-parse jsonv-fuzz-roundtrip
                      PROPERTIES
                        INTERPROCEDURAL_OPTIMIZATION                FALSE
                        INTERPROCEDURAL_OPTIMIZATION_RELEASE        FALSE
                        INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO FALSE
                     )

# Seeds. The JSON_Checker suite and the naughty-string list are already in the repository as test data, so they are
# copied rather than duplicated. canada.json and citm_catalog.json are excluded on purpose: at 2.2 MB and 1.7 MB they
# would dominate the corpus and cripple the iteration rate. Seeds with no equivalent in the test data are committed
# under corpus/ and generated by make_seeds.py.
#
# Copying at configure time rather than from a custom target means the corpus is always there after a build, instead
# of only when someone remembers to name an extra target. The cost is a re-configure to pick up edited seeds. The
# destination is the project's own build root -- build/fuzz-corpus for a top-level build -- which is what fuzz.yml
# and CONTRIBUTING.md name.
file(GLOB JSONV_FUZZ_SEEDS
     "${PROJECT_SOURCE_DIR}/src/jsonv-tests/data/json_checker/*.json"
     "${PROJECT_SOURCE_DIR}/src/jsonv-tests/data/blns.json"
     "${PROJECT_SOURCE_DIR}/src/jsonv-tests/data/generated.json"
     "${PROJECT_SOURCE_DIR}/src/jsonv-tests/data/paths.json"
     "${CMAKE_CURRENT_SOURCE_DIR}/corpus/*.json"
    )
file(COPY ${JSONV_FUZZ_SEEDS} DESTINATION "${PROJECT_BINARY_DIR}/fuzz-corpus")
list(LENGTH JSONV_FUZZ_SEEDS JSONV_FUZZ_SEED_COUNT)
message(STATUS "Fuzz seed corpus: ${JSONV_FUZZ_SEED_COUNT} seeds in ${PROJECT_BINARY_DIR}/fuzz-corpus")
